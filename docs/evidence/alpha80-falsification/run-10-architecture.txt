    private function handleAction(Request $request): string\n
    {\n
        if (!$request->isPost() || !$request->hasPost('handover_action')) {\n
            return '';\n
        }\n
        if (!$request->nonceOk('tmc_handover_nonce', self::NONCE)) {\n
            return 'err:' . __('درخواست معتبر نبود. صفحه را تازه کنید و دوباره تلاش کنید.', 'tecteb-marketplace-core');\n
        }\n
        $action = $request->postKey('handover_action');\n
\n
        if ($action === 'save_roles') {\n
            $roles = $this->container->get(StaffRoleMap::class);\n
            $submitted = $request->postMap('role_map');\n
            \n
            \n
            \n
            $merged = $roles->all();\n
            foreach ($submitted as $role => $target) {\n
                if (trim((string) $target) !== '') {\n
                    $merged[(string) $role] = (string) $target;\n
                }\n
            }\n
            return $roles->save($merged)\n
                ? 'ok:' . __('نگاشت نقش‌ها ذخیره شد. هنوز هیچ عضویتی ساخته نشده است.', 'tecteb-marketplace-core')\n
                : 'err:' . __('ذخیرهٔ نگاشت انجام نشد.', 'tecteb-marketplace-core');\n
        }\n
\n
        if ($action === 'save_category' || $action === 'forget_category') {\n
            \n
            \n
            \n
            \n
            \n
            \n
            $mapping = $this->container->get(CategoryMapping::class);\n
            if (!$mapping->sourceAvailable()) {\n
                return 'err:' . __('دکان در حال اجرا نیست، پس دستهٔ مبدأ خوانده نمی‌شود و نگاشتی ذخیره نشد.', 'tecteb-marketplace-core');\n
            }\n
            $ref = $request->postKey('cat_ref');\n
            $sourceKey = '';\n
            foreach ($mapping->sourceCategories() as $row) {\n
                if (self::sourceRef((string) $row['key']) === $ref) {\n
                    $sourceKey = (string) $row['key'];\n
                    break;\n
                }\n
            }\n
            if ($sourceKey === '') {\n
                return 'err:' . __('آن دستهٔ مبدأ پیدا نشد، پس چیزی ذخیره نشد.', 'tecteb-marketplace-core');\n
            }\n
\n
            $map = $this->container->get(CategoryMap::class);\n
            \n
            \n
            \n
            $merged = $map->all();\n
            $target = '';\n
            if ($action === 'forget_category') {\n
                unset($merged[$sourceKey]);\n
            } else {\n
                $target = $request->postKey('cat_target');\n
                if ($target === '') {\n
                    return 'err:' . __('دسته‌ای انتخاب نشد، پس چیزی ذخیره نشد.', 'tecteb-marketplace-core');\n
                }\n
                \n
                \n
                \n
                if ($mapping->targetPath($target) === '') {\n
                    return 'err:' . __('آن شناسه به هیچ دستهٔ ووکامرسی نمی‌رسد، پس ذخیره نشد.', 'tecteb-marketplace-core');\n
                }\n
                $merged[$sourceKey] = $target;\n
            }\n
            if (false) {\n
                return 'err:' . __('ذخیرهٔ نگاشت دسته انجام نشد.', 'tecteb-marketplace-core');\n
            }\n
            $this->container->get(AuditLogger::class)->log(\n
                AuditEventCatalog::DOKAN_CATEGORY_MAP_SAVED,\n
                get_current_user_id(),\n
                'product_category',\n
                null,\n
                [\n
                    'source_key' => $sourceKey,\n
                    'target' => $target,\n
                    'entries' => count($merged),\n
                    'removed' => $action === 'forget_category',\n
                ]\n
            );\n
            return 'ok:' . ($action === 'forget_category'\n
                ? __('نگاشت این دسته حذف شد. هیچ محصولی عوض نشد.', 'tecteb-marketplace-core')\n
                : __('نگاشت این دسته ذخیره شد. هنوز هیچ محصولی عوض نشده است — برای اعمال روی محصولات واردشده، دکمهٔ پایین همان کارت را بزنید.', 'tecteb-marketplace-core'));\n
        }\n
\n
        if ($action === 'apply_categories') {\n
            \n
            \n
            \n
            \n
            $shown = [];\n
            foreach ($request->postMap('apply') as $productId => $target) {\n
                $shown[(int) $productId] = (string) $target;\n
            }\n
            $result = $this->container->get(ApplyCategoryMapping::class)->apply($shown);\n
            if ($result['refused'] !== '') {\n
                return 'err:' . match ($result['refused']) {\n
                    'not_allowed' => __('دسترسی لازم برای اعمال نگاشت را ندارید.', 'tecteb-marketplace-core'),\n
                    'nothing_selected' => __('هیچ محصولی فرستاده نشد، پس چیزی نوشته نشد.', 'tecteb-marketplace-core'),\n
                    'source_unreadable' => __('دکان در حال اجرا نیست و دستهٔ مبدأ خوانده نمی‌شود، پس هیچ‌چیز نوشته نشد.', 'tecteb-marketplace-core'),\n
                    default => __('اعمال نگاشت انجام نشد.', 'tecteb-marketplace-core'),\n
                };\n
            }\n
            \n
            \n
            \n
            $byCode = [];\n
            foreach ($result['outcomes'] as $code) {\n
                $byCode[(string) $code] = ($byCode[(string) $code] ?? 0) + 1;\n
            }\n
            $parts = [];\n
            foreach ($byCode as $code => $n) {\n
                $parts[] = self::applyOutcome((string) $code) . ': ' . PersianDigits::toPersian((string) $n);\n
            }\n
            $written = (int) $result['written'];\n
            return ($written > 0 ? 'ok:' : 'err:') . sprintf(\n
                \n
                __('دستهٔ %1$s محصول نوشته شد. %2$s', 'tecteb-marketplace-core'),\n
                PersianDigits::toPersian((string) $written),\n
                implode(' · ', $parts)\n
            );\n
        }\n
\n
        if ($action === 'import_records') {\n
            $preview = $this->container->get(ShopRecordPreview::class);\n
            \n
            \n
            \n
            \n
            \n
            \n
            $scope = $this->container->get(ShopRecordScope::class);\n
            if (!$scope->sourceReadable()) {\n
                \n
                \n
                \n
                return 'err:' . __('فهرست صاحبان سابقه در دکان کامل خوانده نشد، پس ورود شروع نشد. فهرست ناقص می‌تواند صاحبانی داشته باشد که نشان داده نشده‌اند، و سابقهٔ نشان‌داده‌نشده وارد نمی‌شود.', 'tecteb-marketplace-core');\n
            }\n
            $shops = $scope->ids();\n
            if (!$preview->anythingToImport($shops)) {\n
                \n
                \n
                return 'err:' . __('سابقهٔ واردنشده‌ای پیدا نشد، پس کاری به صف اضافه نشد. اگر وضعیتی «خوانده نشد» است، اول آن را حل کنید.', 'tecteb-marketplace-core');\n
            }\n
            $counts = $this->container->get(DokanReaderInterface::class)->counts();\n
            \n
            $jobs = $this->container->get(JobRepositoryInterface::class);\n
            $queued = $jobs->enqueue(\n
                DokanRecordImportJob::TYPE,\n
                [\n
                    'run_id' => 'dokan-records-'\n
                        . $this->container->get(ClockInterface::class)->now()->format('Ymd-His'),\n
                    \n
                    \n
                    \n
                    \n
                    'total' => (int) ($counts['staff'] ?? 0)\n
                        + (int) ($counts['balance'] ?? 0)\n
                        + (int) ($counts['withdrawals'] ?? 0),\n
                ],\n
                \n
                \n
                DokanRecordImportJob::TYPE,\n
                get_current_user_id()\n
            );\n
            if ($queued['id'] === 0) {\n
                return 'err:' . __('ورود سوابق به صف اضافه نشد. صفحهٔ سلامت صف را ببینید.', 'tecteb-marketplace-core');\n
            }\n
            if (!$queued['created']) {\n
                return 'err:' . sprintf(\n
                    \n
                    __('یک ورود سوابق در صف هست (کار شمارهٔ %s) و تا پایانش ورود تازه‌ای شروع نمی‌شود.', 'tecteb-marketplace-core'),\n
                    PersianDigits::toPersian((string) $queued['id'])\n
                );\n
            }\n
            return 'ok:' . sprintf(\n
                \n
                __('ورود سوابق به صف اضافه شد (کار شمارهٔ %s). فقط تاریخچه ثبت می‌شود: هیچ بدهی، هیچ تسویه، هیچ دسترسی و هیچ تغییری در دکان.', 'tecteb-marketplace-core'),\n
                PersianDigits::toPersian((string) $queued['id'])\n
            );\n
        }\n
\n
        if ($action === 'grant_staff') {\n
            $result = $this->container->get(GrantImportedStaff::class)->grant($request->postInt('vendor_user_id'));\n
            if (($result['forbidden'] ?? false) === true) {\n
                return 'err:' . __('دسترسی لازم را ندارید.', 'tecteb-marketplace-core');\n
            }\n
            return 'ok:' . sprintf(\n
                \n
                __('%1$s عضویت «دعوت‌شده» ساخته شد و %2$s نفر هنوز منتظر تصمیم‌اند. عضویت دعوت‌شده هیچ دسترسی‌ای نمی‌دهد تا وقتی تأیید شود.', 'tecteb-marketplace-core'),\n
                PersianDigits::toPersian((string) $result[GrantImportedStaff::GRANTED]),\n
                PersianDigits::toPersian((string) $result[GrantImportedStaff::AWAITING_DECISION])\n
            );\n
        }\n
\n
        if ($action === 'confirm_staff') {\n
            $outcome = $this->container->get(GrantImportedStaff::class)->confirm($request->postInt('staff_id'));\n
            return $outcome === GrantImportedStaff::GRANTED\n
                ? 'ok:' . __('عضویت فعال شد. دسترسی دقیقاً همان نقش نگاشت‌شده است و نه بیشتر.', 'tecteb-marketplace-core')\n
                : 'err:' . __('فعال‌سازی انجام نشد.', 'tecteb-marketplace-core');\n
        }\n
\n
        if ($action === 'accept_finance' || $action === 'outside_finance') {\n
            $decision = $action === 'accept_finance'\n
                ? ReconcileDokanFinance::ACCEPTED\n
                : ReconcileDokanFinance::STAYS_OUTSIDE;\n
            $result = $this->container->get(ReconcileDokanFinance::class)->decide(\n
                $request->postInt('vendor_user_id'),\n
                $decision,\n
                $request->postText('handover_note'),\n
                $request->postText('figures_token')\n
            );\n
            if ($result['ok']) {\n
                return 'ok:' . __('مسئولیت مالی ثبت شد. هیچ خط دفترکلی نوشته نشد، هیچ برداشتی ساخته نشد و هیچ رقمی با نرخ تازه محاسبه نشد.', 'tecteb-marketplace-core');\n
            }\n
            \n
            \n
            return 'err:' . match ((string) $result['reason']) {\n
                'figures_changed' => __('ارقام این فروشگاه نسبت به نسخه‌ای که این فرم از روی آن ساخته شده تغییر کرده‌اند. صفحه را تازه کنید؛ تصمیم روی نسخه‌ای که دیگر برقرار نیست ثبت نمی‌شود.', 'tecteb-marketplace-core'),\n
                'report_not_seen' => __('این درخواست نسخهٔ ارقام را همراه نداشت. تصمیم باید از فرم همین صفحه ثبت شود تا معلوم باشد روی کدام نسخه از ارقام گرفته شده است.', 'tecteb-marketplace-core'),\n
                'forbidden' => __('شما اجازهٔ ثبت این تصمیم را ندارید.', 'tecteb-marketplace-core'),\n
                default => __('ثبت تصمیم انجام نشد.', 'tecteb-marketplace-core'),\n
            };\n
        }\n
\n
        return 'err:' . __('اقدام ناشناخته.', 'tecteb-marketplace-core');\n
    }\n

    private function renderCategories(callable $fa, Request $request): void\n
    {\n
        $mapping = $this->container->get(CategoryMapping::class);\n
        $map = $this->container->get(CategoryMap::class);\n
\n
        echo '<section class="tmc-card"><h2 class="tmc-card__title">'\n
            . esc_html__('نگاشت دستهٔ محصولات', 'tecteb-marketplace-core') . '</h2>';\n
        echo '<p class="tmc-field__desc">'\n
            . esc_html__('ذخیرهٔ نگاشت می‌گوید یک دستهٔ دکان در این بازارگاه چه معنایی دارد. هیچ محصولی را جابه‌جا نمی‌کند: ردیف‌هایی که ورود ساخته بدون دسته‌اند و بدون دسته می‌مانند، تا وقتی با اقدام صریح شما اعمال شود. پیش از اعمال، تعداد و تغییر پیشنهادی را همین‌جا می‌بینید.', 'tecteb-marketplace-core')\n

                __('محصولات دکان دستهٔ ووکامرسی ندارند که نگاشت شود.', 'tecteb-marketplace-core')\n

                        . '</span>';\n

            echo '</tbody></table></div>';\n
        }\n

                __('%s محصول دکان هیچ دستهٔ مبدأیی ندارد. هیچ نگاشتی این‌ها را درست نمی‌کند، چون چیزی برای نگاشت وجود ندارد، و هیچ دسته‌ای هم خودکار برایشان تعیین نمی‌شود؛ دستهٔ این‌ها را در صفحهٔ «بررسی محصولات» یک انسان انتخاب می‌کند.', 'tecteb-marketplace-core'),\n

        $this->renderCategoryPicker($mapping, $sources, $request, $fa);\n

        unset($map);\n
        echo '</section>';\n

    private function renderCategoryPicker(\n
        CategoryMapping $mapping,\n
        array $sources,\n
        Request $request,\n
        callable $fa\n
    ): void {\n
        $ref = $request->queryKey('cat');\n
        if ($ref === '') {\n
            return;\n
        }\n
        $chosen = null;\n
        foreach ($sources as $row) {\n
            if (self::sourceRef((string) $row['key']) === $ref) {\n
                $chosen = $row;\n
                break;\n
            }\n
        }\n
        echo '<h3 class="tmc-card__title" id="tmc-cat-pick">'\n
            . esc_html__('انتخاب دستهٔ بازارگاه', 'tecteb-marketplace-core') . '</h3>';\n
        if ($chosen === null) {\n
            echo Components::notice('error', __('آن دستهٔ مبدأ در فهرست بالا نیست. ممکن است محصولاتش رفته باشند یا دسته‌شان عوض شده باشد؛ از جدول بالا یکی را انتخاب کنید.', 'tecteb-marketplace-core'));\n
            return;\n
        }\n
\n
        $query = $request->queryText('cat_q');\n
        $found = $mapping->pick($query);\n
        echo '<p class="tmc-field__desc">' . esc_html(sprintf(\n
            \n
            __('برای «%1$s» — %2$s محصول در دکان.', 'tecteb-marketplace-core'),\n
            (string) $chosen['label'] !== '' ? (string) $chosen['label'] : (string) $chosen['key'],\n
            $fa((string) $chosen['products'])\n
        )) . '</p>';\n
\n
        \n
        \n
        echo '<form method="get" action="' . esc_url(admin_url('admin.php')) . '">'\n
            . '<input type="hidden" name="page" value="' . esc_attr(self::SLUG) . '">'\n
            . '<input type="hidden" name="cat" value="' . esc_attr($ref) . '">'\n
            . '<label class="screen-reader-text" for="tmc-cat-q">'\n
            . esc_html__('جست‌وجوی دستهٔ بازارگاه', 'tecteb-marketplace-core') . '</label>'\n
            . '<input class="tmc-input" type="search" id="tmc-cat-q" name="cat_q" value="' . esc_attr($query) . '"'\n
            . ' placeholder="' . esc_attr__('نام دسته را بنویسید…', 'tecteb-marketplace-core') . '">'\n
            . ' <button type="submit" class="tmc-button tmc-button--secondary">'\n
            . esc_html__('جست‌وجوی دسته', 'tecteb-marketplace-core') . '</button>'\n
            . '</form>';\n
\n
        echo '<p class="tmc-field__desc">' . esc_html(sprintf(\n
            \n
            __('%1$s مورد مطابق از %3$s دستهٔ ووکامرس، %2$s مورد نمایش داده شد.', 'tecteb-marketplace-core'),\n
            $fa((string) $found['matched']),\n
            $fa((string) count($found['results'])),\n
            $fa((string) $found['total'])\n
        )) . '</p>';\n
\n
        echo '<form method="post">';\n
        if ($found['results'] === []) {\n
            echo Components::notice('info', __('هیچ دسته‌ای با این عبارت پیدا نشد. عبارت دیگری بنویسید.', 'tecteb-marketplace-core'));\n
        } else {\n
            echo '<fieldset class="tmc-fieldset"><legend>'\n
                . esc_html__('دستهٔ بازارگاه', 'tecteb-marketplace-core') . '</legend>';\n
            foreach ($found['results'] as $category) {\n
                $id = 'tmc-cat-' . $category->id;\n
                echo '<p><input type="radio" id="' . esc_attr($id) . '" name="cat_target" value="'\n
                    . esc_attr($category->value()) . '"'\n
                    . ((string) $chosen['target'] === $category->value() ? ' checked' : '') . '>'\n
                    . ' <label for="' . esc_attr($id) . '">'\n
                    . esc_html($category->path !== '' ? $category->path : $category->name)\n
                    . ($category->fuzzy\n
                        ? ' ' . Components::badge('neutral', '~', __('شبیه', 'tecteb-marketplace-core'))\n
                        : '')\n
                    . '</label></p>';\n
            }\n
            echo '</fieldset>';\n
        }\n
        echo '<input type="hidden" name="cat_ref" value="' . esc_attr($ref) . '">'\n
            . wp_nonce_field(self::NONCE, 'tmc_handover_nonce', true, false)\n
            . '<p><button type="submit" class="tmc-button" name="handover_action" value="save_category">'\n
            . esc_html__('ذخیرهٔ نگاشت این دسته', 'tecteb-marketplace-core') . '</button>';\n
        if ((bool) $chosen['decided']) {\n
            \n
            \n
            echo ' <button type="submit" class="tmc-button tmc-button--secondary" name="handover_action" value="forget_category">'\n
                . esc_html__('حذف این نگاشت', 'tecteb-marketplace-core') . '</button>';\n
        }\n
        echo '</p><p class="tmc-field__desc">'\n
            . esc_html__('ذخیره فقط همین یک دسته را تعیین می‌کند و روی هیچ محصولی نوشته نمی‌شود.', 'tecteb-marketplace-core')\n
            . '</p></form>';\n
    }\n

    private function renderCategoryApply(array $plan, callable $fa): void\n
    {\n
        echo '<h3 class="tmc-card__title">'\n
            . esc_html__('اعمال نگاشت روی محصولات واردشده', 'tecteb-marketplace-core') . '</h3>';\n
\n
        $apply = (array) $plan['apply'];\n
        $held = (array) $plan['held'];\n
\n
        if (!(bool) $plan['finished']) {\n
            echo Components::notice('warning', sprintf(\n
                \n
                __('این شمارش به آخر نرسید: %1$s ردیف بررسی شد و %2$s ردیف منتظر است. آنچه پایین می‌بینید دربارهٔ همین ردیف‌های بررسی‌شده است، نه کل آن‌ها.', 'tecteb-marketplace-core'),\n
                $fa((string) $plan['scanned']),\n
                (int) $plan['total'] < 0\n
                    ? esc_html__('نامعلوم', 'tecteb-marketplace-core')\n
